SORTING ALGORITHMS
Bubble Sort
procedure bubbleSort(arr, n)
    for i = 0 to n-2 do
        for j = 0 to n-i-2 do
            if arr[j] > arr[j+1] then
                swap(arr[j], arr[j+1])
            end if
        end for
    end for
end procedure
⏱️ O(n²) worst 🔄 Stable 💾 O(1) space
Selection Sort
procedure selectionSort(arr, n)
    for i = 0 to n-2 do
        minIdx = i
        for j = i+1 to n-1 do
            if arr[j] < arr[minIdx] then
                minIdx = j
            end if
        end for
        swap(arr[i], arr[minIdx])
    end for
end procedure
📉 O(n²) 🔁 Unstable
Insertion Sort
procedure insertionSort(arr, n)
    for i = 1 to n-1 do
        key = arr[i]
        j = i-1
        while j >= 0 and arr[j] > key do
            arr[j+1] = arr[j]
            j = j-1
        end while
        arr[j+1] = key
    end for
end procedure
🟢 O(n) best 📦 Adaptive
Merge Sort
procedure mergeSort(arr, left, right)
    if left < right then
        mid = (left+right)/2
        mergeSort(arr, left, mid)
        mergeSort(arr, mid+1, right)
        merge(arr, left, mid, right)
    end if
end procedure
⭐ O(n log n) 📦 O(n) extra
Quick Sort
procedure quickSort(arr, low, high)
    if low < high then
        pi = partition(arr, low, high)
        quickSort(arr, low, pi-1)
        quickSort(arr, pi+1, high)
    end if
end procedure
⚡ avg O(n log n) ⚠️ worst O(n²)
SEARCHING ALGORITHMS
Linear Search
procedure linearSearch(arr, target)
    for i = 0 to length(arr)-1 do
        if arr[i] == target then return i
    end for
    return -1
end procedure
📊 O(n) ✅ unsorted OK
Binary Search
procedure binarySearch(arr, target)
    left = 0, right = length(arr)-1
    while left ≤ right do
        mid = (left+right)//2
        if arr[mid] == target then return mid
        if arr[mid] < target then left = mid+1
        else right = mid-1
    end while
    return -1
end procedure
📈 O(log n) 🔢 requires sorted
Jump Search
procedure jumpSearch(arr, target)
    n = length(arr), step = floor(√n), prev = 0
    while arr[min(step,n)-1] < target do
        prev = step; step += floor(√n)
        if prev ≥ n then return -1
    end while
    for i = prev to min(step,n)-1 do
        if arr[i] == target then return i
    end for
    return -1
end procedure
⏩ O(√n) 📐 sorted array
Interpolation Search
procedure interpolationSearch(arr, target)
    low=0, high=len(arr)-1
    while low ≤ high and target between arr[low] and arr[high] do
        pos = low + ((target-arr[low])*(high-low))/(arr[high]-arr[low])
        if arr[pos] == target then return pos
        if arr[pos] < target then low = pos+1 else high = pos-1
    end while
    return -1
end procedure
✨ O(log log n) avg
GRAPH & PATHFINDING
BFS (Breadth‑First)
procedure BFS(graph, start, goal)
    queue = [start], visited = {start}
    while queue not empty do
        node = queue.dequeue()
        if node == goal then return path
        for neighbor in graph[node] do
            if neighbor not visited then
                visited.add(neighbor); queue.enqueue(neighbor)
            end if
        end for
    end while
    return failure
end procedure
⏱️ O(V+E) 🎯 shortest path (unweighted)
DFS (Depth‑First)
procedure DFS(graph, node, visited, goal)
    if node == goal then return true
    visited.add(node)
    for neighbor in graph[node] do
        if neighbor not visited then
            if DFS(graph, neighbor, visited, goal) then return true
        end if
    end for
    return false
end procedure
🧠 O(V+E) 📚 recursion / stack
Dijkstra (Shortest Path)
procedure dijkstra(graph, source)
    dist[source]=0, PQ = priority queue
    PQ.push((0, source))
    while PQ not empty do
        (d,u) = PQ.extractMin()
        if d > dist[u] then continue
        for each neighbor v with weight w do
            alt = dist[u] + w
            if alt < dist[v] then
                dist[v] = alt; PQ.push((alt, v))
            end if
        end for
    end while
    return dist
end procedure
⏱️ O((V+E) log V) 🎯 non‑negative weights
A* Search
procedure AStar(start, goal, h)
    openSet = {start}, g[start]=0, f[start]=h(start,goal)
    while openSet not empty do
        current = node in openSet with min f
        if current == goal then reconstruct path
        openSet.remove(current)
        for neighbor of current do
            tent_g = g[current] + d(current, neighbor)
            if tent_g < g[neighbor] then
                cameFrom[neighbor]=current
                g[neighbor]=tent_g
                f[neighbor]=g[neighbor]+h(neighbor,goal)
                openSet.add(neighbor)
            end if
        end for
    end while
    return failure
end procedure
🎯 optimal + heuristic
